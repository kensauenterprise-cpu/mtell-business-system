// Check if the products table has an is_deal column
$hasDealColumn = false;

$check = $conn->query("SHOW COLUMNS FROM products LIKE 'is_deal'");

if ($check && $check->num_rows > 0) {
    $hasDealColumn = true;
}

// Build SQL
if ($hasDealColumn) {

    $sql = "
    SELECT
        id,
        name,
        online_price,
        retail_price,
        online_image,
        image,
        stock
    FROM products
    WHERE
        category_id = 1
        AND online_price > 0
        AND stock > 0
        AND is_deal = 1
        AND (
            online_image <> ''
            OR image <> ''
        )
    ORDER BY created_at DESC
    LIMIT 8
    ";

} else {

    $sql = "
    SELECT
        id,
        name,
        online_price,
        retail_price,
        online_image,
        image,
        stock
    FROM products
    WHERE
        category_id = 1
        AND online_price > 0
        AND stock > 0
        AND (
            online_image <> ''
            OR image <> ''
        )
    ORDER BY created_at DESC
    LIMIT 8
    ";

}

$result = $conn->query($sql);

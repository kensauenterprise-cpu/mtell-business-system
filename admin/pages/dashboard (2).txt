// ==========================
// ROLE PERMISSIONS
// ==========================
function canAccess($tab, $role) {

    global $all_tabs;

    $permissions = [

        'super_admin' => $all_tabs,
        'admin' => $all_tabs,
        'it' => $all_tabs,

        'manager' => [
            'home','dashboard','orders','inventory','financials',
            'reports','customers','suppliers','expenses'
        ],

        'cashier' => [
            'home','orders','products','pos','transactions'
        ]
    ];

    return in_array($tab, $permissions[$role] ?? []);
}

// ==========================
// SAFE INCLUDE
// ==========================
function safeInclude($file) {

    global $conn;

    if (!isset($conn)) {
        return "<p style='color:red'>❌ Database connection missing</p>";
    }

    if (!file_exists($file)) {
        return "<p style='color:red'>❌ File not found: $file</p>";
    }

    ob_start();
    include $file;
    return ob_get_clean();
}

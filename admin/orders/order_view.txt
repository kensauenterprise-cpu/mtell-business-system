<?php
require_once $_SERVER['DOCUMENT_ROOT'].'/infinity/admin/includes/init.php';
require_once $_SERVER['DOCUMENT_ROOT'].'/infinity/admin/includes/role.php';

// ==========================
// 🔐 SESSION
// ==========================
if (session_status() === PHP_SESSION_NONE) {
    session_start();
}

// ==========================
// 🔥 DB
// ==========================
$conn = $GLOBALS['conn'] ?? null;

if (!$conn) {
    die("❌ DB connection missing");
}

// ==========================
// 🏢 BRANCH
// ==========================
$branch_id = (int)($_SESSION['branch_id'] ?? 1);

// ==========================
// 📦 ORDER ID
// ==========================
$order_id = (int)($_GET['id'] ?? 0);

if ($order_id <= 0) {
    die("❌ Invalid order ID");
}

// ==========================
// 💰 RECEIVE PAYMENT
// ==========================
if (isset($_POST['receive_payment'])) {

    $payment_method = trim(
        $_POST['payment_method'] ?? 'cash'
    );

    $stmtPay = $conn->prepare("
        UPDATE orders
        SET
            payment_method = ?,
            payment_status = 'paid'
        WHERE id = ?
        AND branch_id = ?
    ");

    if ($stmtPay) {

        $stmtPay->bind_param(
            "sii",
            $payment_method,
            $order_id,
            $branch_id
        );

        $stmtPay->execute();

        $stmtPay->close();
    }

    header("Location: order_view.php?id=".$order_id);
    exit;
}

// ==========================
// 🚚 MARK DELIVERED
// ==========================
if (isset($_POST['mark_delivered'])) {

    $stmtDelivered = $conn->prepare("
        UPDATE orders
        SET status = 'Delivered'
        WHERE id = ?
        AND branch_id = ?
    ");

    if ($stmtDelivered) {

        $stmtDelivered->bind_param(
            "ii",
            $order_id,
            $branch_id
        );

        $stmtDelivered->execute();

        $stmtDelivered->close();
    }

    header("Location: order_view.php?id=".$order_id);
    exit;
}

// ==========================
// 📦 ORDER HEADER
// ==========================
$stmt = $conn->prepare("
    SELECT *
    FROM orders
    WHERE id=?
    AND branch_id=?
");

$stmt->bind_param(
    "ii",
    $order_id,
    $branch_id
);

$stmt->execute();

$order = $stmt->get_result()->fetch_assoc();

if (!$order) {
    die("❌ Order not found");
}

// ==========================
// 🧾 ORDER ITEMS
// ==========================
$items = $conn->prepare("
    SELECT oi.*, p.name, p.cost_price
    FROM order_items oi
    LEFT JOIN products p ON p.id = oi.product_id
    WHERE oi.order_id=?
");

$items->bind_param("i", $order_id);

$items->execute();

$items = $items->get_result();

// ==========================
// 🚚 RIDER (OPTIONAL)
// ==========================
$rider = null;

if (!empty($order['rider_id'])) {

    $r = $conn->query("
        SELECT *
        FROM riders
        WHERE id=".(int)$order['rider_id']
    );

    $rider = $r
        ? $r->fetch_assoc()
        : null;
}

// ==========================
// 💰 CALCULATIONS
// ==========================
$total_cost = 0;
$total_profit = 0;
?>

            <?= ($order['status'] ?? '')=='Delivered'

<!-- ========================== -->

<?php if ($items && $items->num_rows > 0): ?>

    <?php while($i = $items->fetch_assoc()):

    $qty = $i['qty'] ?? 1;

    $sale = $i['price'] ?? 0;

    $cost = $i['cost_price'] ?? 0;

    $profit = ($sale - $cost) * $qty;

    $total_cost += $cost * $qty;

    $total_profit += $profit;
    ?>

    <tr>

        <td>
            <?= htmlspecialchars($i['name'] ?? '') ?>
        </td>

        <td>
            <?= (int)$qty ?>
        </td>

        <td>
            KES <?= number_format($sale,2) ?>
        </td>

        <td>
            KES <?= number_format($cost,2) ?>
        </td>

        <td style="color:<?= $profit>=0?'green':'red' ?>">

            KES <?= number_format($profit,2) ?>

        </td>

    </tr>

    <?php endwhile; ?>

<?php else: ?>

<tr>

    <td colspan="5">

        No item records found

    </td>

</tr>

<?php endif; ?>

</table>

<!-- ========================== -->

    </p>

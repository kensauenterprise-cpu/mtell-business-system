function sendMailMessage($toEmail,$toName,$subject,$body){

    $mail = new PHPMailer(true);

    try{

        $mail->isSMTP();

        $mail->Host = 'smtp.gmail.com';

        $mail->SMTPAuth = true;

        $mail->Username = 'kensauenterprise@gmail.com';

        /*
        Replace this with your Gmail App Password
        */
        $mail->Password = 'YOUR_GMAIL_APP_PASSWORD';

        $mail->SMTPSecure = PHPMailer::ENCRYPTION_STARTTLS;

        $mail->Port = 587;

        $mail->setFrom(
            'kensauenterprise@gmail.com',
            'Mtell Online Shopping'
        );

        $mail->addAddress($toEmail,$toName);

        $mail->isHTML(true);

        $mail->Subject = $subject;

        $mail->Body = $body;

        return $mail->send();

    }catch(Exception $e){

        error_log($mail->ErrorInfo);

        return false;

    }

}
